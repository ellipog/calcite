# Redstone that runs by itself (CAL-116), in the flat world's chunk (0, 0) by its spawn, on the
# grass, whose top is at y -61. It's built with ticks frozen, so the ticks each block set schedules
# are saved, and run once a player's in the chunk. A lamp's set before what powers it, so it's lit
# as that comes; and the second observer of each clock's set last, so the first sees it come and
# starts the clock.
setworldspawn 0 -60 0

# Two observers watching each other are a clock, each lighting a lamp behind it.
setblock 1 -60 2 minecraft:redstone_lamp
setblock 4 -60 2 minecraft:redstone_lamp
setblock 2 -60 2 minecraft:observer[facing=east]
setblock 3 -60 2 minecraft:observer[facing=west]

# A clock's pulses along wire, through a repeater, into a lamp.
fill 4 -60 5 7 -60 5 minecraft:redstone_wire
setblock 8 -60 5 minecraft:repeater[facing=west,delay=1]
fill 9 -60 5 10 -60 5 minecraft:redstone_wire
setblock 11 -60 5 minecraft:redstone_lamp
setblock 2 -60 5 minecraft:observer[facing=east]
setblock 3 -60 5 minecraft:observer[facing=west]

# A torch powering the stone it's on, through wire round a corner and back into the stone, off
# and on every 2 ticks till it burns out, and on again 160 ticks after. An observer's pulse is too
# short for a torch: it's off again before the torch's tick.
setblock 12 -60 8 minecraft:stone
setblock 12 -60 9 minecraft:redstone_wall_torch[facing=south]
setblock 12 -60 10 minecraft:redstone_wire
setblock 13 -60 10 minecraft:redstone_wire
setblock 14 -60 10 minecraft:redstone_wire
setblock 14 -60 9 minecraft:redstone_wire
setblock 14 -60 8 minecraft:redstone_wire
setblock 13 -60 8 minecraft:redstone_wire

# A redstone block powering wire, once, into a repeater of the longest delay, which lights a lamp.
fill 2 -60 11 5 -60 11 minecraft:redstone_wire
setblock 6 -60 11 minecraft:repeater[facing=west,delay=4]
setblock 7 -60 11 minecraft:redstone_lamp
setblock 1 -60 11 minecraft:redstone_block
