# Armor stands (CAL-155), in the flat world's chunk (0, 0) by its spawn, on a stone floor, which
# both servers load and tick as the session watches: falling 10 blocks onto stone, a hay bale, a
# honey block and a slime block; sliding east on ice; down a ladder; into a pool of water; and,
# standing, a small one turned, a marker, one without gravity in the air, one posed with its arms
# and no base plate, and an invisible one.
setworldspawn 0 -60 0
fill 1 -61 1 14 -61 14 minecraft:stone

# Falling onto stone, hay, honey and slime, which it bounces on.
summon minecraft:armor_stand 2.5 -50 2.5
setblock 4 -61 2 minecraft:hay_block
summon minecraft:armor_stand 4.5 -50 2.5
setblock 6 -61 2 minecraft:honey_block
summon minecraft:armor_stand 6.5 -50 2.5
setblock 8 -61 2 minecraft:slime_block
summon minecraft:armor_stand 8.5 -50 2.5

# Sliding east on ice from 0.6 a tick, onto the stone past it.
fill 2 -61 5 8 -61 5 minecraft:ice
summon minecraft:armor_stand 2.5 -60 5.5 {Motion:[0.6d,0.0d,0.0d]}

# Down a ladder on a stone wall from 7 up.
fill 12 -60 4 12 -52 4 minecraft:stone
fill 12 -60 5 12 -53 5 minecraft:ladder[facing=south]
summon minecraft:armor_stand 12.5 -53 5.5

# Into a pool of water three deep, from 6 up.
fill 3 -63 10 5 -61 12 minecraft:water
summon minecraft:armor_stand 4.5 -55 11.5

# Standing: small and turned; a marker; without gravity, 3 up; posed, with its arms and no base
# plate; and invisible.
summon minecraft:armor_stand 8.5 -60 9.5 {Small:1b,Rotation:[45.0f,0.0f]}
summon minecraft:armor_stand 10.5 -60 9.5 {Marker:1b}
summon minecraft:armor_stand 12.5 -57 9.5 {NoGravity:1b,Rotation:[-90.0f,0.0f]}
summon minecraft:armor_stand 8.5 -60 12.5 {ShowArms:1b,NoBasePlate:1b,Pose:{Head:[10.0f,20.0f,0.0f],RightArm:[-60.0f,0.0f,10.0f]}}
summon minecraft:armor_stand 10.5 -60 12.5 {Invisible:1b}
