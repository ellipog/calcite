# Beacons (D520), on the flat world's grass in chunk (0, 0), on normal at noon, each with the
# effects it was set with: one of jump boost with regeneration besides on two levels of gold,
# which gives only its jump boost, with red stained glass over it, on which the bot puts stone;
# one of speed on a 3 by 3 of iron, one level, a corner of which the bot breaks; one of
# resistance on emerald, under bedrock, which lets its beam through; and, side by side on four
# levels of diamond, one of strength with regeneration and one of haste twice over, haste II. All
# are in one chunk, as the order a level's block entities tick in follows the order their chunks
# were loaded in, which isn't the same on each server. As in `armadillo.txt`, the chunks within
# the simulation distance of 3 of the spawn but (0, 0) are forced, so both servers tick the same
# chunks from the start, and the beacons, which only tick in a ticking chunk, pulse alike.
setworldspawn 5 -60 5
difficulty normal
time set 6000
forceload add -48 -48 63 63
fill 0 -61 0 4 -61 4 minecraft:gold_block
fill 1 -60 1 3 -60 3 minecraft:gold_block
setblock 2 -59 2 minecraft:beacon{primary_effect:"minecraft:jump_boost",secondary_effect:"minecraft:regeneration"}
setblock 2 -58 2 minecraft:red_stained_glass
fill 1 -61 7 3 -61 9 minecraft:iron_block
setblock 2 -60 8 minecraft:beacon{primary_effect:"minecraft:speed"}
fill 1 -61 12 3 -61 14 minecraft:emerald_block
setblock 2 -60 13 minecraft:beacon{primary_effect:"minecraft:resistance"}
setblock 2 -59 13 minecraft:bedrock
fill 6 -60 6 15 -60 14 minecraft:diamond_block
fill 7 -59 7 14 -59 13 minecraft:diamond_block
fill 8 -58 8 13 -58 12 minecraft:diamond_block
fill 9 -57 9 12 -57 11 minecraft:diamond_block
setblock 10 -56 10 minecraft:beacon{primary_effect:"minecraft:strength",secondary_effect:"minecraft:regeneration"}
setblock 11 -56 10 minecraft:beacon{primary_effect:"minecraft:haste",secondary_effect:"minecraft:haste"}
forceload remove 0 0
