# Blocks to use items on (CAL-142), on the flat world's grass in chunk (0, 0), by its spawn, within
# a survival player's reach of it: an empty composter, an empty flower pot, a lit candle, a cake, an
# empty cauldron, a full water cauldron and a lava cauldron; and a lectern, a chiseled bookshelf
# and a decorated pot.
setworldspawn 0 -60 0
setblock 2 -60 0 minecraft:composter
setblock -2 -60 0 minecraft:flower_pot
setblock -2 -60 2 minecraft:candle[lit=true]
setblock 0 -60 2 minecraft:cake
setblock 0 -60 -2 minecraft:cauldron
setblock 2 -60 -2 minecraft:water_cauldron[level=3]
setblock 2 -60 2 minecraft:lava_cauldron
# For `use-books`: a lectern, a chiseled bookshelf facing the spawn and a decorated pot.
setblock -2 -60 -2 minecraft:lectern[facing=south]
setblock 0 -60 4 minecraft:chiseled_bookshelf[facing=north]
setblock -3 -60 -1 minecraft:decorated_pot
