# Pistons (CAL-125), in the flat world's chunk (0, 0) by its spawn, on a stone floor, as the
# redstone fixture lays one. Block events aren't saved and don't run while the build's ticks are
# frozen, so each piston here moves only once a tick that waits in the save powers it: an observer
# clock's, or a repeater's, set before the redstone block behind it. Nothing pops, as what a
# broken block drops comes with items.
setworldspawn 0 -60 0
fill 0 -61 0 15 -61 15 minecraft:stone

# Observer clocks, each pair's second powering the piston behind it with a pulse of two ticks:
# a piston pushing stone out and drawing its head back; a sticky one, whose pulses are too short
# for it to pull the stone back before it's placed; and a sticky one facing up under slime, with
# stone on the slime's side that it carries along and honey on its other side that slime doesn't
# stick to.
setblock 4 -60 1 minecraft:stone
setblock 3 -60 1 minecraft:piston[facing=east]
setblock 1 -60 1 minecraft:observer[facing=east]
setblock 2 -60 1 minecraft:observer[facing=west]
setblock 4 -60 4 minecraft:stone
setblock 3 -60 4 minecraft:sticky_piston[facing=east]
setblock 1 -60 4 minecraft:observer[facing=east]
setblock 2 -60 4 minecraft:observer[facing=west]
setblock 9 -59 1 minecraft:slime_block
setblock 10 -59 1 minecraft:stone
setblock 9 -59 2 minecraft:honey_block
setblock 9 -60 1 minecraft:sticky_piston[facing=up]
setblock 7 -60 1 minecraft:observer[facing=east]
setblock 8 -60 1 minecraft:observer[facing=west]

# Repeaters, each set before the redstone block it reads, so its tick waits in the save, then
# powering a piston for good: one pushing twelve stones, as many as it can; one in front of
# thirteen, which stays in; a sticky one in front of obsidian, which stays in; and one pushing a
# redstone block up to a lamp, which lights.
setblock 3 -60 8 minecraft:stone
setblock 4 -60 8 minecraft:stone
setblock 5 -60 8 minecraft:stone
setblock 6 -60 8 minecraft:stone
setblock 7 -60 8 minecraft:stone
setblock 8 -60 8 minecraft:stone
setblock 9 -60 8 minecraft:stone
setblock 10 -60 8 minecraft:stone
setblock 11 -60 8 minecraft:stone
setblock 12 -60 8 minecraft:stone
setblock 13 -60 8 minecraft:stone
setblock 14 -60 8 minecraft:stone
setblock 2 -60 8 minecraft:piston[facing=east]
setblock 1 -60 8 minecraft:repeater[facing=west,delay=2]
setblock 0 -60 8 minecraft:redstone_block
fill 3 -60 10 15 -60 10 minecraft:stone
setblock 2 -60 10 minecraft:piston[facing=east]
setblock 1 -60 10 minecraft:repeater[facing=west,delay=2]
setblock 0 -60 10 minecraft:redstone_block
setblock 3 -60 12 minecraft:obsidian
setblock 2 -60 12 minecraft:sticky_piston[facing=east]
setblock 1 -60 12 minecraft:repeater[facing=west,delay=2]
setblock 0 -60 12 minecraft:redstone_block
setblock 5 -60 14 minecraft:redstone_lamp
setblock 3 -60 14 minecraft:redstone_block
setblock 2 -60 14 minecraft:piston[facing=east]
setblock 1 -60 14 minecraft:repeater[facing=west,delay=4]
setblock 0 -60 14 minecraft:redstone_block
