# Villagers throwing each other what they have too much of (CAL-160), on the flat world's grass in
# the morning, when their schedules have them idle: Ada, a nitwit, with 30 bread and 40 wheat
# seeds, and Bo, a farmer by his composter, with 50 wheat and no food, three blocks apart, who go
# to each other and trade, `TradeWithVillager`: Ada throws Bo the bread she has more than 24 of, as
# he wants food, and half her seeds, which a farmer asks for and a nitwit doesn't; Bo, a farmer,
# throws her half his wheat. Neither may pick things up, as a summoned mob's loaded, so what they
# throw stays where it falls. Bo's saved memories hold his job site, and a nitwit takes none, so
# neither looks for one. Summoned with their data, so nothing's drawn for their spawn. Natural
# spawning's off in the session. As in `villagers.txt`, the chunks within the simulation distance
# of 3 of the spawn but the villagers' are forced, so both servers tick the same chunks from the
# bot's join and their draws line up; both are in the one chunk, (0, 0), so both servers read
# them, and give them their IDs and places in the tick, in the order they were saved.
setworldspawn 0 -60 0
time set 1000
forceload add -48 -48 63 63
setblock 10 -60 7 minecraft:composter
summon minecraft:villager 6.5 -60 4.5 {PersistenceRequired:1b,CustomName:"Ada",VillagerData:{type:"minecraft:plains",profession:"minecraft:nitwit",level:1},Inventory:[{id:"minecraft:bread",count:30},{id:"minecraft:wheat_seeds",count:40}],Rotation:[270.0f,0.0f]}
summon minecraft:villager 9.5 -60 4.5 {PersistenceRequired:1b,CustomName:"Bo",Xp:10,VillagerData:{type:"minecraft:plains",profession:"minecraft:farmer",level:2},Inventory:[{id:"minecraft:wheat",count:50}],Brain:{memories:{"minecraft:job_site":{value:{pos:[I;10,-60,7],dimension:"minecraft:overworld"}}}},Rotation:[90.0f,0.0f]}
forceload remove 0 0
