# Blocks mobs stand on, `stepOn` (D431), in the flat world's chunk (0, 0) by the spawn, each mob
# penned in glass two high: a cow on magma, which burns it every 10 ticks till it dies; a pig on
# redstone ore, which it lights, the ore's particles drawn from the level's random each tick; and
# a sheep on a floor of slime blocks, which slow it as it wanders.
setworldspawn 0 -60 0
fill 3 -60 0 5 -59 2 minecraft:glass
fill 4 -60 1 4 -59 1 minecraft:air
setblock 4 -61 1 minecraft:magma_block
summon minecraft:cow 4.5 -60 1.5 {PersistenceRequired:1b}
fill 7 -60 0 9 -59 2 minecraft:glass
fill 8 -60 1 8 -59 1 minecraft:air
setblock 8 -61 1 minecraft:redstone_ore
summon minecraft:pig 8.5 -60 1.5 {PersistenceRequired:1b}
fill 3 -60 4 7 -59 8 minecraft:glass
fill 4 -60 5 6 -59 7 minecraft:air
fill 4 -61 5 6 -61 7 minecraft:slime_block
summon minecraft:sheep 5.5 -60 6.5 {PersistenceRequired:1b}
