# Villagers hiding as a bell rings (CAL-160), on the flat world's grass in chunk (0, 0) in the
# morning, when their schedules have grown villagers with no job site idle: a bell on the floor two
# blocks east of the bot's spawn, facing north, which the bot rings at 200; and Ada and Bo, grown
# with no profession, in a pen walled two high in stone east of it with two beds. Each claims a bed
# as its home, `AcquirePoi`, and the bell as its meeting point. Within 32 of the bell as it rings,
# both hear it, `HEARD_BELL_TIME`, and hide, `ReactToBell`: each picks a bed to hide by,
# `LocateHidingPlace`, by its own random where it isn't by one, and walks there, counting the ticks
# it's by it, `SetHiddenState`, till 300 ticks after the ring, when its schedule has it idle again.
# Summoned with their data, so nothing's drawn for their spawn. Natural spawning's off in the
# session. As in `villagers.txt`, the chunks within the simulation distance of 3 of the spawn but
# (0, 0) are forced, so both servers tick the same chunks from the bot's join and their draws line
# up.
setworldspawn 0 -60 0
time set 1000
forceload add -48 -48 63 63
setblock 2 -60 0 minecraft:bell[attachment=floor,facing=north]
fill 4 -60 -3 14 -59 -3 minecraft:stone
fill 4 -60 11 14 -59 11 minecraft:stone
fill 4 -60 -2 4 -59 10 minecraft:stone
fill 14 -60 -2 14 -59 10 minecraft:stone
setblock 7 -60 9 minecraft:white_bed[facing=north,part=foot]
setblock 7 -60 8 minecraft:white_bed[facing=north,part=head]
setblock 11 -60 9 minecraft:red_bed[facing=north,part=foot]
setblock 11 -60 8 minecraft:red_bed[facing=north,part=head]
summon minecraft:villager 6.5 -60 1.5 {PersistenceRequired:1b,CustomName:"Ada",Rotation:[270.0f,0.0f]}
summon minecraft:villager 11.5 -60 1.5 {PersistenceRequired:1b,CustomName:"Bo",Rotation:[90.0f,0.0f]}
forceload remove 0 0
