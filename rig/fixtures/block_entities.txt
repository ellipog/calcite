# Block entities (CAL-115), in the flat world's chunk (0, 0) by its spawn, on a stone floor, as
# the redstone fixture lays one. Each chunk packet sends the ones saved here with their update
# tags, and two heads behind an observer clock are powered and unpowered at each pulse, each
# sent with its update packet as it changes. None of them ticks into anything a client's sent,
# or looks for players: no spawners, vaults, trial spawners, end gateways or creaking hearts.
setworldspawn 0 -60 0
fill 0 -61 0 15 -61 15 minecraft:stone

# Those whose update tag is all their own: a sign with text, a hanging sign under stone, a skull,
# a decorated pot with sherds, a jigsaw, a structure block, a beacon and a conduit, alone.
setblock 1 -60 1 minecraft:oak_sign[rotation=4]{front_text:{messages:["Calcite","","",""],color:"blue",has_glowing_text:1b},is_waxed:1b}
setblock 3 -59 1 minecraft:stone
setblock 3 -60 1 minecraft:oak_hanging_sign[rotation=8]{back_text:{messages:["","under","stone",""]}}
setblock 5 -60 1 minecraft:skeleton_skull[rotation=2]
setblock 7 -60 1 minecraft:decorated_pot{sherds:["minecraft:brick","minecraft:arms_up_pottery_sherd","minecraft:brick","minecraft:skull_pottery_sherd"]}
setblock 9 -60 1 minecraft:jigsaw
setblock 11 -60 1 minecraft:structure_block[mode=save]{name:"calcite:test",sizeX:3,sizeY:3,sizeZ:3}
setblock 13 -60 1 minecraft:beacon
setblock 15 -60 1 minecraft:conduit[waterlogged=false]

# A banner, with its components too; a campfire, out, with its items; a shelf with its items;
# a copper golem statue, whose update tag is empty; and a chest, which sends none.
setblock 1 -60 4 minecraft:white_banner[rotation=12]{patterns:[{color:"red",pattern:"minecraft:stripe_bottom"},{color:"blue",pattern:"minecraft:cross"}]}
setblock 3 -60 4 minecraft:campfire[lit=false]{Items:[{Slot:0b,id:"minecraft:beef",count:1}]}
setblock 5 -60 4 minecraft:oak_shelf[facing=south]{Items:[{Slot:1b,id:"minecraft:apple",count:2}]}
setblock 7 -60 4 minecraft:copper_golem_statue
setblock 9 -60 4 minecraft:chest{Items:[{Slot:0b,id:"minecraft:diamond",count:3}]}

# Two observers watching each other are a clock, each powering a head behind it, a dragon's and a
# piglin's, which move while they're powered. The second observer's set last, so the first sees
# it come and starts the clock.
setblock 1 -60 8 minecraft:dragon_head[rotation=4]
setblock 4 -60 8 minecraft:piglin_head[rotation=12]
setblock 2 -60 8 minecraft:observer[facing=east]
setblock 3 -60 8 minecraft:observer[facing=west]
