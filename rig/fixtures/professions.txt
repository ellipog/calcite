# Villagers taking and losing professions by their job sites (CAL-160), on the flat world's grass
# in the morning, when their schedules have them idle: Cy, grown with no profession, by a
# composter; Di, a farmer with no job site, by a fletching table; and Ed and Fa, farmers whose
# saved memories hold the same composter as their job site, Ed with more experience. Di loses her
# profession and takes the fletching table as a fletcher; Fa forgets the composter Ed holds and
# loses her profession, takes it again and loses it again, then walks to Cy's composter and takes
# it, a farmer there for good, her brain made again each time; and Cy, whose first look for a job
# site finds none he can reach, is left with none. Summoned with their data, so nothing's drawn
# for their spawn. Natural spawning's off in the session. As in `villagers.txt`, the chunks within
# the simulation distance of 3 of the spawn but the villagers' are forced, so both servers tick
# the same chunks from the bot's join and their draws line up; all four are in the one chunk,
# (0, 0), so both servers read them, and give them their IDs and places in the tick, in the order
# they were saved.
setworldspawn 0 -60 0
time set 1000
forceload add -48 -48 63 63
setblock 8 -60 1 minecraft:composter
setblock 3 -60 13 minecraft:fletching_table
setblock 14 -60 12 minecraft:composter
summon minecraft:villager 6.5 -60 1.5 {PersistenceRequired:1b,CustomName:"Cy",Rotation:[90.0f,0.0f]}
summon minecraft:villager 1.5 -60 13.5 {PersistenceRequired:1b,CustomName:"Di",VillagerData:{type:"minecraft:plains",profession:"minecraft:farmer",level:1},Rotation:[90.0f,0.0f]}
summon minecraft:villager 14.5 -60 10.5 {PersistenceRequired:1b,CustomName:"Ed",Xp:10,VillagerData:{type:"minecraft:plains",profession:"minecraft:farmer",level:2},Brain:{memories:{"minecraft:job_site":{value:{pos:[I;14,-60,12],dimension:"minecraft:overworld"}}}},Rotation:[180.0f,0.0f]}
summon minecraft:villager 14.5 -60 14.5 {PersistenceRequired:1b,CustomName:"Fa",VillagerData:{type:"minecraft:plains",profession:"minecraft:farmer",level:1},Brain:{memories:{"minecraft:job_site":{value:{pos:[I;14,-60,12],dimension:"minecraft:overworld"}}}},Rotation:[0.0f,0.0f]}
forceload remove 0 0
