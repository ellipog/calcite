# Pens for armadillos whose chunks stop ticking their entities and start again (CAL-184), on the
# flat world's grass: one in chunk (0, 0) by the bot's spawn, and one in chunk (2, 0), 40 blocks
# east, each keeping what's in it in its chunk. `requeue`'s steps summon the armadillos once the
# bot's in, so they start to tick in the order they're summoned on both servers, whatever order
# each loads the chunks about the bot in. The chunks within 8 of the spawn are forced, so wherever
# the bot goes its view has no chunk to load, but the pens' two, which tick their entities only
# with the bot's view about them, its loading tickets' level 31; next to the forced ones, they
# tick their blocks but not their entities while it's away.
setworldspawn 0 -60 0
forceload add -128 -128 127 127
fill -1 -60 -1 3 -60 -1 minecraft:oak_fence
fill -1 -60 3 3 -60 3 minecraft:oak_fence
fill -1 -60 0 -1 -60 2 minecraft:oak_fence
fill 3 -60 0 3 -60 2 minecraft:oak_fence
fill 38 -60 -1 42 -60 -1 minecraft:oak_fence
fill 38 -60 3 42 -60 3 minecraft:oak_fence
fill 38 -60 0 38 -60 2 minecraft:oak_fence
fill 42 -60 0 42 -60 2 minecraft:oak_fence
forceload remove 0 0
forceload remove 32 0
