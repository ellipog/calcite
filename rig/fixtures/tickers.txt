# Block entities' tickers (CAL-130), in the flat world's chunk (0, 0) by its spawn, on a stone
# floor, as the receivers fixture lays one. The time's set to dusk, so the sky darkens and the
# sun goes down as the session watches, and the daylight detectors' signals change with them.
setworldspawn 0 -60 0
fill 0 -61 0 15 -61 15 minecraft:stone
time set 11950

# Daylight detectors: under the open sky, plain and inverted, each lighting a lamp beside it;
# under leaves; under a roof of stone, lit from the sides; boxed in stone, inverted, which
# puts out 15 once it first ticks; and one whose signal runs down wire to a lamp.
setblock 1 -60 1 minecraft:daylight_detector
setblock 2 -60 1 minecraft:redstone_lamp
setblock 4 -60 1 minecraft:daylight_detector[inverted=true]
setblock 5 -60 1 minecraft:redstone_lamp
setblock 7 -60 1 minecraft:daylight_detector
setblock 7 -58 1 minecraft:oak_leaves[persistent=true]
fill 9 -58 0 11 -58 2 minecraft:stone
setblock 10 -60 1 minecraft:daylight_detector
fill 12 -60 0 14 -59 2 minecraft:stone
setblock 13 -60 1 minecraft:daylight_detector[inverted=true]
setblock 1 -60 4 minecraft:daylight_detector
setblock 2 -60 4 minecraft:redstone_wire
setblock 3 -60 4 minecraft:redstone_wire
setblock 4 -60 4 minecraft:redstone_lamp

# Counted down as they tick: a crafter crafting with 3 ticks to go, which stops; a brewing
# stand showing a bottle it hasn't, which shows none once it first ticks; and an empty hopper,
# whose cooldown runs out.
setblock 7 -60 4 minecraft:crafter[crafting=true]{crafting_ticks_remaining:3}
setblock 9 -60 4 minecraft:brewing_stand[has_bottle_1=true]
setblock 11 -60 4 minecraft:hopper

# Tickers with nothing to do, which do nothing: signs, a bell, a shulker box, an empty
# furnace, campfires lit and not, a beacon on stone, a dry conduit, an empty hive, a sculk
# catalyst, and a chest, whose ticker is the client's alone.
setblock 1 -60 8 minecraft:oak_sign
setblock 3 -60 8 minecraft:oak_wall_hanging_sign[facing=south]
setblock 5 -60 8 minecraft:bell[attachment=floor,facing=north]
setblock 7 -60 8 minecraft:shulker_box
setblock 9 -60 8 minecraft:furnace
setblock 11 -60 8 minecraft:campfire
setblock 13 -60 8 minecraft:campfire[lit=false]
setblock 1 -60 11 minecraft:beacon
setblock 3 -60 11 minecraft:conduit[waterlogged=false]
setblock 5 -60 11 minecraft:beehive[facing=south]
setblock 7 -60 11 minecraft:sculk_catalyst
setblock 9 -60 11 minecraft:chest[facing=south]

# Jukeboxes playing cat's record (CAL-143), each beside a lamp set lit, which it powers while it
# plays: one from its song's start, a note every 20 ticks of a color from the level's random, and
# one 20 ticks from its song's end, which stops and tells its lamp, which goes out.
setblock 1 -60 14 minecraft:jukebox[has_record=true]{RecordItem:{id:"minecraft:music_disc_cat",count:1},ticks_since_song_started:0L}
setblock 2 -60 14 minecraft:redstone_lamp[lit=true]
setblock 4 -60 14 minecraft:jukebox[has_record=true]{RecordItem:{id:"minecraft:music_disc_cat",count:1},ticks_since_song_started:3700L}
setblock 5 -60 14 minecraft:redstone_lamp[lit=true]
