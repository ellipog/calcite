# floor, as the receivers fixture lays one.
setworldspawn 0 -60 0
fill 0 -61 0 15 -61 15 minecraft:stone

# A lake two deep in a stone basin, 9 by 9, with a sponge set in the middle of its top as the
# world's built, which takes up as much of it as a sponge set by a command does. As the world runs,
# the rest flows into the hole.

fill 2 -61 2 10 -60 10 minecraft:water
setblock 6 -60 6 minecraft:sponge

# A channel along z at x 14, walled in stone, with a dry sponge at its north end and a water
# source at its south end, which flows up to the sponge as the world runs and is taken up.
fill 13 -60 1 15 -60 8 minecraft:stone
fill 14 -60 2 14 -60 7 minecraft:air
setblock 14 -60 2 minecraft:sponge
setblock 14 -60 7 minecraft:water

# Frosted ice in daylight, in stone rings so what it melts into stays: one alone at x 2, and a
# square of four at x 6 and 7, filled at once so none melts as the others are set. Each ages as its
# tick runs and melts at age 3, and the square's around one that melts age with it.
