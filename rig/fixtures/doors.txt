# Villagers opening and shutting a door (CAL-160), on the flat world in the morning, when their
# schedules have them idle: Cy and Di, grown with no profession, shut in a cobblestone hut with
# a floor and a roof, its inside three across and two high, whose one way out is an oak door in
# its east wall; outside it a composter and a fletching table. Each takes one as the job site it
# found, walks out to it through the door, opening it as its path comes to it and shutting it
# behind it once it's passed, but not while the other's coming through, and takes it as its job
# site. The door's halves are set lower first, so neither breaks the other. Summoned with their
# data, so nothing's drawn for their spawn, and falling as fast as a tick's gravity has them, so
# their first move lands them and their first looks for a job site find a path: one off the
# ground finds none, and puts both sites off in an order vanilla's identity hashes set. Natural
# spawning's off in the session. As in `villagers.txt`, the chunks within the simulation distance
# of 3 of the spawn but (0, 0), the villagers', are forced, so both servers tick the same chunks
# from the bot's join and their draws line up.
setworldspawn 0 -60 0
time set 1000
forceload add -48 -48 63 63
fill 4 -61 4 8 -58 8 minecraft:cobblestone hollow
setblock 8 -60 6 minecraft:oak_door[facing=east,half=lower,hinge=left,open=false]
setblock 8 -59 6 minecraft:oak_door[facing=east,half=upper,hinge=left,open=false]
setblock 15 -60 6 minecraft:composter
setblock 15 -60 2 minecraft:fletching_table
summon minecraft:villager 6.5 -60 6.5 {PersistenceRequired:1b,OnGround:1b,Motion:[0.0d,-0.0784000015258789d,0.0d],CustomName:"Cy",Rotation:[270.0f,0.0f]}
summon minecraft:villager 5.5 -60 5.5 {PersistenceRequired:1b,OnGround:1b,Motion:[0.0d,-0.0784000015258789d,0.0d],CustomName:"Di",Rotation:[270.0f,0.0f]}
forceload remove 0 0
