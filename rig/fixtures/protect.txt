# Zombies in enchanted armor (D439), on the flat world's grass in chunk (0, 0), east of its spawn
# within a survival player's reach, on normal, which keeps monsters, each without AI in an oak
# fence cell of its own under a stone roof, so the sun doesn't burn them: one in a diamond
# chestplate with protection IV; one in diamond boots with frost walker II and one in a leather
# helmet with fire protection IV, which give nothing against an attack. The last two stand on
# magma, which doesn't burn them, as a mob without AI doesn't move and `stepOn` comes with a
# move. Natural spawning's off in the session, so nothing else comes.

difficulty normal
fill 1 -60 -4 3 -60 -4 minecraft:oak_fence
fill 1 -60 -2 3 -60 -2 minecraft:oak_fence
fill 1 -60 0 3 -60 0 minecraft:oak_fence

fill 1 -60 -3 1 -60 1 minecraft:oak_fence

setblock 2 -61 -1 minecraft:magma_block

summon minecraft:zombie 2.5 -60 -2.5 {NoAI:1b,PersistenceRequired:1b,equipment:{chest:{id:"minecraft:diamond_chestplate",count:1,components:{"minecraft:enchantments":{"minecraft:protection":4}}}}}
summon minecraft:zombie 2.5 -60 -0.5 {NoAI:1b,PersistenceRequired:1b,equipment:{feet:{id:"minecraft:diamond_boots",count:1,components:{"minecraft:enchantments":{"minecraft:frost_walker":2}}}}}
summon minecraft:zombie 2.5 -60 1.5 {NoAI:1b,PersistenceRequired:1b,equipment:{head:{id:"minecraft:leather_helmet",count:1,components:{"minecraft:enchantments":{"minecraft:fire_protection":4}}}}}
